cmake_minimum_required(VERSION 3.22.1)
project(talkme_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=2)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
    message(FATAL_ERROR "Missing verified whisper.cpp source. Run prepareWhisperSource before configuring CMake.")
endif()
add_subdirectory("${WHISPER_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/whisper")
add_library(talkme_whisper SHARED whisper_jni.cpp)
target_link_libraries(talkme_whisper PRIVATE whisper android log)
target_compile_options(talkme_whisper PRIVATE -O3)
target_link_options(talkme_whisper PRIVATE "-Wl,-z,max-page-size=16384")
